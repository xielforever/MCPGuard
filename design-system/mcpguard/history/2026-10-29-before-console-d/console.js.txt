'use strict';
const $=id=>document.getElementById(id);
let state,csrf='',page='操作与审批',detail;
function node(tag,text,cls){const n=document.createElement(tag);if(text!==undefined)n.textContent=text;if(cls)n.className=cls;return n}
function message(s,error=false){$('message').textContent=s;$('message').className='status'+(error?' error':'')}
async function api(path,options={}){const r=await fetch('/console/'+path,{credentials:'same-origin',...options,headers:{'Content-Type':'application/json','X-CSRF-Token':csrf,...options.headers}});if(!r.ok){if(r.status===401)showLogin();throw new Error(await r.text())}return r.json()}
function showLogin(){$('login').hidden=false;$('workspace').hidden=true;$('logout').hidden=true;csrf='';state=null}
function button(text,fn,cls){const b=node('button',text,cls);b.type='button';b.addEventListener('click',()=>Promise.resolve().then(fn).catch(e=>message(e.message,true)));return b}
function table(parent,headers,rows){const t=node('table'),head=node('tr');for(const h of headers)head.append(node('th',h));t.append(head);for(const row of rows){const tr=node('tr');for(const cell of row){const td=node('td');td.append(cell instanceof Node?cell:node('span',String(cell??'')));tr.append(td)}t.append(tr)}parent.append(t)}
function field(form,label,kind='text',options=[]){const l=node('label',label);let input;if(kind==='select'){input=node('select');for(const value of options){const o=node('option',typeof value==='string'?value:value.label);o.value=typeof value==='string'?value:value.value;input.append(o)}}else{input=node(kind==='textarea'?'textarea':'input');input.type=kind}l.append(input);form.append(l);return input}
function choiceUsers(){return(state.users||[]).map(u=>({value:u.id,label:u.name+' · '+u.scopeId}))}
async function refresh(){state=await api('api/state');csrf=state.csrf;$('login').hidden=true;$('workspace').hidden=false;$('logout').hidden=false;$('identity').textContent=(state.admin?'管理员':'当前身份')+' · '+state.scopeId;$('version').textContent='配置版本 '+state.revision;const nav=$('nav');nav.replaceChildren();for(const p of ['操作与审批','资源','规则','用户与权限','本人审计','通知与设置'])nav.append(button(p,()=>{page=p;detail=null;render()}));render()}
async function configure(payload){await api('api/config',{method:'PATCH',headers:{'If-Match':'"'+state.revision+'"','Idempotency-Key':crypto.randomUUID()},body:JSON.stringify(payload)});message('已保存。旧审批计划在配置变化后需要重新申请。');await refresh()}
function render(){const c=$('content');c.replaceChildren(node('h2',page));if(page==='操作与审批'){if(detail){renderDetail(c);return}table(c,['工具','状态','到期时间','操作'],state.actions.map(a=>[a.toolName,a.state,new Date(a.expiresAt).toLocaleString(),button('查看计划',async()=>{detail=await api('api/actions/'+a.actionId);render()})]));if(!state.actions.length)c.append(node('p','暂无可见操作。'))}
 if(page==='资源'){c.append(node('p','只显示资源绑定与配置状态；连接地址和凭据由本机私有配置管理。','muted'));table(c,['资源','连接配置','状态','操作'],state.resources.map(r=>[r.name,r.configured?'已配置':'未配置',r.enabled?'启用':'停用',state.admin?button(r.enabled?'停用':'启用',()=>configure({kind:'resource',resource:r.name,enabled:!r.enabled})):'' ]))}
 if(page==='规则'){c.append(node('p','拒绝规则优先。受控写入始终需要他人审批；ALLOW 不能授予缺失的工具权限。','muted'));if(!state.admin){c.append(node('p','规则管理需要管理员权限。'));return}table(c,['工具','效果','匹配表达式','状态'],state.rules.map(r=>[r.toolName,r.effect,r.pattern,r.enabled?'启用':'停用']));const f=node('form');f.append(node('h3','新增规则'));const tool=field(f,'工具','select',['pg.update_demo_note','redis.get_demo_note','redis.set_demo_note','directory.get_demo_office','inventory.reserve_demo']);const effect=field(f,'效果','select',['DENY','REQUIRE_APPROVAL','ALLOW']);const pattern=field(f,'参数匹配表达式');f.append(button('保存规则',()=>configure({kind:'rule',id:crypto.randomUUID(),toolName:tool.value,effect:effect.value,pattern:pattern.value,enabled:true})));c.append(f);for(const r of state.rules)c.append(button((r.enabled?'停用 ':'启用 ')+r.toolName+' '+r.id.slice(0,8),()=>configure({kind:'rule',id:r.id,toolName:r.toolName,effect:r.effect,pattern:r.pattern,enabled:!r.enabled})))}
 if(page==='用户与权限'){if(!state.admin){c.append(node('p','用户与权限管理需要管理员权限。'));return}table(c,['用户','数据范围','状态','业务权限','管理权限'],state.users.map(u=>[u.name,u.scopeId,u.enabled?'有效':'停用',u.canRead?'有效':'无',u.admin?'管理员':'普通身份']));const f=node('form');f.append(node('h3','调整现有身份'));const who=field(f,'用户','select',choiceUsers());const scope=field(f,'数据范围');const enabled=field(f,'身份有效','checkbox');enabled.checked=true;const canRead=field(f,'业务权限有效','checkbox');canRead.checked=true;f.append(button('保存身份',()=>configure({kind:'principal',principalId:who.value,scopeId:scope.value,enabled:enabled.checked,canRead:canRead.checked})));c.append(f);const g=node('form');g.append(node('h3','工具及审批授权'));const user=field(g,'用户','select',choiceUsers());const kind=field(g,'权限类型','select',[{value:'grant',label:'业务工具'},{value:'reviewer',label:'资源审批'},{value:'member',label:'控制台管理'}]);const tool=field(g,'业务工具','select',['pg.update_demo_note','redis.get_demo_note','redis.set_demo_note','directory.get_demo_office','inventory.reserve_demo']);const resource=field(g,'审批资源','select',state.resources.map(r=>r.name));const allowed=field(g,'授权','checkbox');g.append(button('保存授权',()=>configure(kind.value==='grant'?{kind:'grant',principalId:user.value,toolName:tool.value,allowed:allowed.checked}:kind.value==='reviewer'?{kind:'reviewer',principalId:user.value,resource:resource.value,allowed:allowed.checked}:{kind:'member',principalId:user.value,admin:allowed.checked})));c.append(g);c.append(node('h3','现有工具授权'));table(c,['用户','工具','授权'],state.grants.map(g=>[state.users.find(u=>u.id===g.principalId)?.name||g.principalId,g.toolName,String(g.allowed)]));c.append(node('h3','现有审批授权'));table(c,['用户','资源','授权'],state.reviewers.map(g=>[state.users.find(u=>u.id===g.principalId)?.name||g.principalId,g.resource,String(g.allowed)]))}
 if(page==='用户与权限'&&state.admin)renderGroups(c);
 if(page==='本人审计'){c.dataset.after='0';c.append(button('读取下一页',async()=>{const data=await api('api/audit?after='+encodeURIComponent(c.dataset.after||'0')+'&limit=25');c.dataset.after=data.nextAfter;table(c,['序号','事件','操作 ID','时间'],data.events.map(e=>[e.seq,e.eventType,e.actionId,new Date(e.createdAt).toLocaleString()]));message('已读取 '+data.events.length+' 条本人事件。')}))}
 if(page==='通知与设置'){c.append(node('p','企业 SSO：待选择身份源。当前为本机身份模式。'));c.append(node('p','Telegram：'+(state.notificationConfigured?'已配置；投递状态如下。':'待配置 Bot 和接收目标。审批无需依赖通知送达。')));c.append(node('p','通知只包含安全摘要和控制台入口，不发送业务参数、登录凭据或审批验证码。','muted'));table(c,['操作 ID','投递状态','尝试次数'],(state.notifications||[]).map(n=>[n.actionId,n.state,n.attempts]))}}
function renderDetail(c){const a=detail;c.append(button('返回列表',()=>{detail=null;render()}));table(c,['项目','内容'],[['操作 ID',a.actionId],['工具',a.toolName],['申请人',a.requesterId],['数据范围',a.scopeId],['状态',a.state],['到期',new Date(a.expiresAt).toLocaleString()]]);c.append(node('h3','冻结的请求参数'),node('pre',JSON.stringify(a.arguments,null,2)));if(a.canReview){c.append(node('p','确认以上目标、参数和数据范围后作出决定。批准不会立即执行。'));const decide=async vote=>{await api('api/actions/'+a.actionId+'/decision',{method:'POST',headers:{'If-Match':'"'+a.revision+'"','Idempotency-Key':crypto.randomUUID()},body:JSON.stringify({vote,planHash:a.planHash})});detail=null;message(vote==='APPROVE'?'已批准，等待申请者显式恢复。':'已拒绝。');await refresh()};c.append(button('批准',()=>decide('APPROVE'),'primary'),document.createTextNode(' '),button('拒绝',()=>decide('REJECT'),'danger'))}else c.append(node('p',a.requesterId===state.principalId?'此为本人申请，不能自行审批。':'当前操作不可审批。','muted'))}
function renderGroups(c){c.append(node('h3','用户组'));table(c,['组名','状态','操作'],(state.groups||[]).map(g=>[g.name,g.enabled?'启用':'停用',button(g.enabled?'停用':'启用',()=>configure({kind:'group',id:g.id,name:g.name,enabled:!g.enabled}))]));const f=node('form');const name=field(f,'新组名');f.append(button('创建用户组',()=>configure({kind:'group',id:crypto.randomUUID(),name:name.value,enabled:false})));c.append(f);if(!state.groups?.length)return;const g=node('form');const group=field(g,'用户组','select',state.groups.map(v=>({value:v.id,label:v.name})));const kind=field(g,'操作类型','select',[{value:'groupMember',label:'组成员'},{value:'groupGrant',label:'组工具权限'},{value:'groupReviewer',label:'组审批权限'}]);const user=field(g,'用户','select',choiceUsers());const tool=field(g,'工具','select',['pg.update_demo_note','redis.get_demo_note','redis.set_demo_note','directory.get_demo_office','inventory.reserve_demo']);const resource=field(g,'资源','select',state.resources.map(r=>r.name));const allowed=field(g,'加入或授权','checkbox');g.append(button('保存组配置',()=>configure(kind.value==='groupMember'?{kind:kind.value,groupId:group.value,principalId:user.value,allowed:allowed.checked}:kind.value==='groupGrant'?{kind:kind.value,groupId:group.value,toolName:tool.value,allowed:allowed.checked}:{kind:kind.value,groupId:group.value,resource:resource.value,allowed:allowed.checked})));c.append(g);table(c,['用户组','成员'],(state.groupMembers||[]).map(m=>[state.groups.find(g=>g.id===m.groupId)?.name,state.users.find(u=>u.id===m.principalId)?.name]));table(c,['用户组','工具或审批资源','授权'],[...(state.groupGrants||[]),...(state.groupReviewers||[])].map(m=>[state.groups.find(g=>g.id===m.groupId)?.name,m.toolName||m.resource,String(m.allowed)]))}
$('loginForm').addEventListener('submit',async e=>{e.preventDefault();try{await api('login',{method:'POST',body:JSON.stringify({username:$('username').value,password:$('password').value})});$('password').value='';await refresh();message('已登录。')}catch(e){message(e.message,true)}});
$('refresh').addEventListener('click',()=>refresh().catch(e=>message(e.message,true)));
$('logout').addEventListener('click',async()=>{try{await api('api/logout',{method:'POST',body:'{}'});showLogin();message('已退出。')}catch(e){message(e.message,true)}});
refresh().then(async()=>{const id=new URLSearchParams(location.hash.slice(1)).get('action');if(id){detail=await api('api/actions/'+encodeURIComponent(id));render()}}).catch(()=>showLogin());
